cmake_minimum_required(VERSION 3.22.1)
project(larpdevkit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(larpdevkit SHARED il2cpp_dump.cpp)
add_library(larpmain SHARED libmain_proxy.cpp)
set_target_properties(larpmain PROPERTIES OUTPUT_NAME main)
find_library(log-lib log)
target_compile_options(larpdevkit PRIVATE -O2 -fno-rtti)
target_compile_options(larpmain PRIVATE -O2 -fno-rtti)
target_link_libraries(larpdevkit ${log-lib})
target_link_libraries(larpmain ${log-lib} dl)
target_link_options(larpdevkit PRIVATE -Wl,-z,max-page-size=16384)
target_link_options(larpmain PRIVATE -Wl,-z,max-page-size=16384)
